424template <typename T>
425QVector<double> convertToDoubleVector(const QVector<T>& inputVector)
426{
427 QVector<double> doubleVector;
428 doubleVector.reserve(inputVector.size()); // Reserve space to avoid multiple allocations
429
430 for (const T& value : inputVector)
431 {
432 if constexpr (std::is_arithmetic_v<T>) // Check if T is an arithmetic type
433 {
434 doubleVector.append(static_cast<double>(value));
435 }
436 else
437 {
438 qDebug() << "Type cannot be converted to double.";
439 }
440 }
441
442 return doubleVector;
443}
